-- Crate File Audit Table
CREATE TABLE controll.FileAudit
(
    FileAuditID        BIGINT IDENTITY(1,1) PRIMARY KEY,
    DatasetName        VARCHAR(100) NOT NULL,
    FileName           VARCHAR(300) NOT NULL,
    FilePath           VARCHAR(500) NOT NULL,
    BatchID             UNIQUEIDENTIFIER NOT NULL,
    PipelineRunID       VARCHAR(100) NULL,

    StartTime           DATETIME2(3) NOT NULL,
    EndTime             DATETIME2(3) NULL,

    Status              VARCHAR(30) NOT NULL,
    RowsCopied         BIGINT NULL,

    ErrorMessage        VARCHAR(MAX) NULL
);

-- Stored Procedure Insert File Audit
CREATE OR ALTER PROCEDURE control.usp_InsertFileAudit
(
    @DatasetName   VARCHAR(100),
    @FileName      VARCHAR(300),
    @FilePath      VARCHAR(500),
    @BatchID       UNIQUEIDENTIFIER,
    @PipelineRunID VARCHAR(100)
)
AS
BEGIN

    SET NOCOUNT ON;

    INSERT INTO control.FileAudit
    (
        DatasetName,
        FileName,
        FilePath,
        BatchID,
        PipelineRunID,
        StartTime,
        Status
    )
    VALUES
    (
        @DatasetName,
        @FileName,
        @FilePath,
        @BatchID,
        @PipelineRunID,
        SYSUTCDATETIME(),
        'Running'
    );

    SELECT CAST(SCOPE_IDENTITY() AS BIGINT) AS FileAuditID;

END;
GO
